"""Synthetic JPEG decode experiment; no GUI or user inputs."""
import os
os.environ['QT_QPA_PLATFORM'] = 'offscreen'
import argparse
from collections import defaultdict
from io import BytesIO
import json
import math
from pathlib import Path
from statistics import median
import sys
from time import perf_counter
from unittest.mock import patch

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from PIL import Image, ImageChops, ImageOps, ImageStat
from app.thumbnail_render import ThumbnailRenderSpec, render_pil_thumbnail, render_owned_jpeg_letterbox
from scripts.benchmark_viewer_navigation import _jpeg_payload


def candidate(image, spec):
    result = render_owned_jpeg_letterbox(image, spec)
    return result if result is not None else render_pil_thumbnail(image, spec)[0]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--mode', choices=['baseline', 'candidate'], required=True)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    payload = _jpeg_payload((2401, 3603), 0, fixture_mode='high-detail')
    spec = ThumbnailRenderSpec(181, 256, 'portrait_1_sqrt2', 'letterbox')
    stages = defaultdict(list)
    # Inclusive method timings; nested load calls must not be summed with copy.
    def timing(cls, name):
        original = getattr(cls, name)
        def wrapped(*a, **k):
            start = perf_counter()
            try:
                return original(*a, **k)
            finally:
                stages[name].append((perf_counter()-start)*1000)
        return patch.object(cls, name, wrapped)
    total = []
    run = candidate if args.mode == 'candidate' else lambda im, sp: render_pil_thumbnail(im, sp)[0]
    from PIL.JpegImagePlugin import JpegImageFile
    with timing(JpegImageFile, 'load'), timing(Image.Image, 'copy'), timing(Image.Image, 'resize'), timing(ImageOps, 'exif_transpose'):
        for _ in range(9):
            start = perf_counter()
            with Image.open(BytesIO(payload)) as image:
                result = run(image, spec)
            total.append((perf_counter()-start)*1000)
    def pixels(qimage):
        rgb = qimage.convertToFormat(qimage.Format.Format_RGB888)
        return Image.frombytes('RGB', (rgb.width(), rgb.height()), bytes(rgb.constBits()), 'raw', 'RGB', rgb.bytesPerLine())
    with Image.open(BytesIO(payload)) as image:
        baseline = pixels(render_pil_thumbnail(image, spec)[0])
    actual = pixels(result)
    difference = ImageStat.Stat(ImageChops.difference(baseline, actual))
    rms = math.sqrt(sum(v*v for v in difference.rms)/3)
    report = dict(mode=args.mode, iterations=9, total_ms=total, median_ms=median(total),
                  stages_inclusive={k: dict(total_ms=sum(v), median_ms=median(v), count=len(v)) for k,v in stages.items()},
                  size=actual.size, mean_absolute_error=sum(difference.mean)/3, psnr_db=20*math.log10(255/rms) if rms else None)
    args.output.write_text(json.dumps(report, indent=2), encoding='utf-8')
    print(json.dumps(report))


if __name__ == '__main__':
    main()
